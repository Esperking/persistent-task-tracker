  <div class="app-container">
    <h2>Persistent Task Tracker</h2>
    
    <div class="input-group">
      <input id="task-input" type="text" placeholder="Enter a task...">
      <button id="add-task-btn">Add</button>
      <button id="clear-btn">Clear All</button>
    </div>

    <ul id="task-list"></ul>
  </div>

  <script>
    // 1. DOM REFERENCES
    const taskInput = document.querySelector('#task-input');
    const addTaskBtn = document.querySelector('#add-task-btn');
    const clearBtn = document.querySelector('#clear-btn');
    const taskList = document.querySelector('#task-list');

    // 2. STATE / DATA
    const savedData = localStorage.getItem('myTasks');
    let tasksArray = savedData ? JSON.parse(savedData) : [];

    // 3. CORE LOGIC / HELPER FUNCTIONS
    function saveAndRender() {
      localStorage.setItem('myTasks', JSON.stringify(tasksArray));
      renderTasks();
    }

    function renderTasks() {
      taskList.innerHTML = ""; // Clear existing UI

      tasksArray.forEach(function(task, index) {
        const newLi = document.createElement('li');
        newLi.textContent = task.text;

        if (task.completed) {
          newLi.classList.add('completed');
        }

        // Toggle completed on row click
        newLi.addEventListener('click', function() {
          tasksArray[index].completed = !tasksArray[index].completed;
          saveAndRender();
        });

        // Delete button
        const deleteBtn = document.createElement('button');
        deleteBtn.textContent = "X";
        deleteBtn.addEventListener('click', function(event) {
          event.stopPropagation();
          tasksArray.splice(index, 1);
          saveAndRender();
        });

        newLi.appendChild(deleteBtn);
        taskList.appendChild(newLi);
      });
    }

    // 4. EVENT LISTENERS
    addTaskBtn.addEventListener('click', function() {
      let userText = taskInput.value;
      if (userText === "") return;

      tasksArray.push({ text: userText, completed: false });
      saveAndRender();
      taskInput.value = "";
    });

    clearBtn.addEventListener('click', function() {
      tasksArray = [];
      saveAndRender();
    });

    // 5. INITIALIZATION
    renderTasks();
  </script>
